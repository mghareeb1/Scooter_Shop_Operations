/** * اسم المنظومة: Scooter Shop Operations API - Full CRUD & Logs * المميزات: جلب، إضافة، تعديل، حذف، وتوثيق سجل التحديثات لحظياً * مربوط بـ: واجهة index.html عبر doGet و doPost */ function doGet(e) { try { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Tickets"); if (!sheet) { return responseJSON({ success: false, message: "تبويب Tickets غير موجود." }); } const data = sheet.getDataRange().getValues(); const tickets = []; // قراءة البيانات من الصف الثاني بعد الهيدر for (let i = 1; i < data.length; i++) { const row = data[i]; if (row[0]) { tickets.push({ id: String(row[0]), date: formatDate(row[1]), branch: String(row[2]), category: String(row[3]), clientName: String(row[4]), clientPhone: String(row[5] || ""), scooterModel: String(row[6]), chassisNo: String(row[7] || "-"), issue: String(row[8]), priority: String(row[9]), assignee: String(row[10]), status: String(row[11]), notes: String(row[12] || "") }); } } return responseJSON({ success: true, tickets: tickets }); } catch (err) { return responseJSON({ success: false, error: err.toString() }); } } function doPost(e) { try { const payload = JSON.parse(e.postData.contents); const action = payload.action; const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheet = ss.getSheetByName("Tickets"); if (!sheet) { return responseJSON({ success: false, message: "تبويب Tickets غير موجود." }); } // 1. فتح تذكرة جديدة if (action === "createTicket") { const lastRow = sheet.getLastRow(); const nextId = "SS-" + (1000 + lastRow); const timestamp = new Date(); sheet.appendRow([ nextId, timestamp, payload.branch, payload.category, payload.clientName, payload.clientPhone || "-", payload.scooterModel, payload.chassisNo || "-", payload.issue, payload.priority || "عادي", payload.assignee || "غير محدد", "جديدة", `[${formatDate(timestamp)}] تم إنشاء التذكرة بواسطة الإدارة` ]); return responseJSON({ success: true, message: "تم إنشاء التذكرة بنجاح", id: nextId }); } // 2. تحديث الحالة وإضافة ملاحظة للسجل if (action === "updateStatus") { const ticketId = payload.id; const newStatus = payload.status; const note = (payload.note || "").trim(); const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(ticketId)) { const rowNum = i + 1; sheet.getRange(rowNum, 12).setValue(newStatus); const currentNotes = String(sheet.getRange(rowNum, 13).getValue() || ""); const timeStr = formatDate(new Date()); const newEntry = note ? `[${timeStr}] (${newStatus}): ${note}` : `[${timeStr}] تم تغيير الحالة إلى (${newStatus})`; const updatedNotes = currentNotes ? currentNotes + "\n" + newEntry : newEntry; sheet.getRange(rowNum, 13).setValue(updatedNotes); return responseJSON({ success: true, message: "تم تحديث التذكرة بنجاح" }); } } return responseJSON({ success: false, message: "التذكرة غير موجودة" }); } // 3. تعديل بيانات التذكرة بالكامل if (action === "editTicket") { const ticketId = payload.id; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(ticketId)) { const rowNum = i + 1; sheet.getRange(rowNum, 3).setValue(payload.branch); sheet.getRange(rowNum, 4).setValue(payload.category); sheet.getRange(rowNum, 5).setValue(payload.clientName); sheet.getRange(rowNum, 6).setValue(payload.clientPhone || "-"); sheet.getRange(rowNum, 7).setValue(payload.scooterModel); sheet.getRange(rowNum, 8).setValue(payload.chassisNo || "-"); sheet.getRange(rowNum, 9).setValue(payload.issue); sheet.getRange(rowNum, 10).setValue(payload.priority); sheet.getRange(rowNum, 11).setValue(payload.assignee); const currentNotes = String(sheet.getRange(rowNum, 13).getValue() || ""); const updatedNotes = currentNotes + "\n" + `[${formatDate(new Date())}] تم تعديل بيانات التذكرة`; sheet.getRange(rowNum, 13).setValue(updatedNotes); return responseJSON({ success: true, message: "تم تعديل بيانات التذكرة بنجاح" }); } } return responseJSON({ success: false, message: "التذكرة غير موجودة" }); } // 4. حذف التذكرة if (action === "deleteTicket") { const ticketId = payload.id; const data = sheet.getDataRange().getValues(); for (let i = 1; i < data.length; i++) { if (String(data[i][0]) === String(ticketId)) { sheet.deleteRow(i + 1); return responseJSON({ success: true, message: "تم حذف التذكرة بنجاح" }); } } return responseJSON({ success: false, message: "التذكرة غير موجودة" }); } return responseJSON({ success: false, message: "إجراء غير معروف" }); } catch (err) { return responseJSON({ success: false, error: err.toString() }); } } function formatDate(date) { if (!date) return ""; try { return Utilities.formatDate(new Date(date), "GMT+2", "yyyy-MM-dd HH:mm"); } catch (e) { return String(date); } } function responseJSON(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
